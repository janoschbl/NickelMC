42/// Number of slots in a shulker box (3 rows of 9).
43pub const SHULKER_BOX_SLOTS: usize = 27;
44const ANIMATION_STEPS: u8 = 10;
45
46/// The current animation state of a shulker box.
47#[derive(Debug, Clone, Copy)]
48pub enum AnimationStatus {
49    /// Fully closed.
50    Closed,
51    /// Opening.
52    Opening,
53    /// Fully open.
54    Opened,
55    /// Closing.
56    Closing,
57}
58
59pub struct ShulkerBoxAnimation {
60    animation_status: AnimationStatus,
61    progress: u8,
62    old_progress: u8,
63}
64
65impl ShulkerBoxAnimation {
66    fn progress(&self) -> f32 {
67        f32::from(self.progress) / f32::from(ANIMATION_STEPS)
68    }
69
70    fn old_progress(&self) -> f32 {
71        f32::from(self.old_progress) / f32::from(ANIMATION_STEPS)
72    }
73}

88fn do_neighbor_updates(world: &Arc<World>, pos: BlockPos, state: BlockStateId) {
89    world.update_neighbour_shapes(state, pos, UpdateFlags::UPDATE_ALL, 512);
90    world.update_neighbors_at(pos, state.get_block());
91}

94    /// Creates a new shulker box block entity.
95    #[must_use]
96    pub fn new(level: Weak<World>, pos: BlockPos, state: BlockStateId) -> Self {
97        Self {
98            storage: RandomizableContainerStorage::new(
99                &vanilla_block_entity_types::SHULKER_BOX,
100                SHULKER_BOX_SLOTS,
101                level,
102                pos,
103                state,
104            ),
105            properties: ContainerProperties::default(),
106            animation: SyncMutex::new(ShulkerBoxAnimation {
107                animation_status: AnimationStatus::Closed,
108                progress: 0,
109                old_progress: 0,
110            }),
111            open_count: AtomicI32::new(0),
112        }
113    }
114
115    fn update_animation(&self, world: &Arc<World>, pos: BlockPos, state: BlockStateId) {
116        let mut animation = self.animation.lock();
117        animation.old_progress = animation.progress;
118        match animation.animation_status {
119            AnimationStatus::Closed => animation.progress = 0,
120            AnimationStatus::Opening => {
121                animation.progress += 1;
122                if animation.old_progress == 0 {
123                    do_neighbor_updates(world, pos, state);
124                }
125
126                if animation.progress >= ANIMATION_STEPS {
127                    animation.animation_status = AnimationStatus::Opened;
128                    animation.progress = ANIMATION_STEPS;
129                    do_neighbor_updates(world, pos, state);
130                }
131
132                drop(animation);
133                self.move_collided_entities(world, pos, state);
134            }
135            AnimationStatus::Opened => animation.progress = ANIMATION_STEPS,
136            AnimationStatus::Closing => {
137                animation.progress = animation.progress.saturating_sub(1);
138                if animation.old_progress == ANIMATION_STEPS {
139                    do_neighbor_updates(world, pos, state);
140                }
141
142                if animation.progress == 0 {
143                    animation.animation_status = AnimationStatus::Closed;
144                    animation.progress = 0;
145                    do_neighbor_updates(world, pos, state);
146                }
147            }
148        }
149    }
150
151    fn move_collided_entities(&self, world: &Arc<World>, pos: BlockPos, state: BlockStateId) {
152        let direction = state.get_value(&BlockStateProperties::FACING);
153
154        let aabb: WorldAabb = {
155            let animation = self.animation.lock();
156            Self::get_progress_delta_aabb(
157                1.0,
158                direction,
159                animation.old_progress(),
160                animation.progress(),
161                DVec3::from(pos.get_bottom_center()),
162            )
163        };
164
165        let entities = world.get_entities_in_aabb(&aabb);
166        for entity in entities {
167            if entity.piston_push_reaction() == PushReaction::IgnoreEntity {
168                continue;
169            }
170            let (offset_x, offset_y, offset_z) = direction.offset();
171            entity.move_entity(
172                MoverType::ShulkerBox,
173                DVec3::new(
174                    (aabb.width() + 0.01) * f64::from(offset_x),
175                    (aabb.height() + 0.01) * f64::from(offset_y),
176                    (aabb.depth() + 0.01) * f64::from(offset_z),
177                ),
178            );
179        }
180    }

182    /// Builds a shulker box item containing the stored inventory.
183    pub fn shulker_box_as_item(&self, state: BlockStateId) -> std::io::Result<ItemStack> {
184        let block_item = REGISTRY.items.by_block(state.get_block());
185        let mut patch = DataComponentPatch::new();
186        for (key, value) in self.collect_components()?.iter() {
187            patch.set_raw(key.clone(), value.clone());
188        }
189        Ok(ItemStack::with_count_and_patch(block_item, 1, patch))
190    }

192    /// Calculates the bounding box of a shulker during its opening animation.
193    ///
194    /// The bounds are expanded in the opening direction based on the animation progress.
195    #[must_use]
196    pub fn get_progress_delta_aabb<I: Space>(
197        size: f32,
198        direction: Direction,
199        progress_from: f32,
200        progress_to: f32,
201        position: DVec3,
202    ) -> Aabb<DVec3, I> {
203        let size = f64::from(size);
204        let bounds =
205            Aabb::<DVec3, I>::new(-size * 0.5, 0.0, -size * 0.5, size * 0.5, size, size * 0.5);
206
207        let max_movement = f64::from(progress_from.max(progress_to));
208        let min_movement = f64::from(progress_from.min(progress_to));
209
210        let dir = DVec3::from(direction.offset_vec());
211
212        bounds
213            .expand_towards(dir * max_movement * size)
214            .contract(-dir * (1.0 + min_movement) * size)
215            .translate(position)
216    }

218    /// Block-local bounds of the box including the lid at its current progress.
219    #[must_use]
220    pub fn get_bounding_box(&self, state: BlockStateId) -> BlockLocalAabb {
221        let bottom_center = DVec3::new(0.5, 0.0, 0.5);
222        Self::get_progress_delta_aabb(
223            1.0,
224            state.get_value(ShulkerBoxBlock::FACING),
225            -1.0,
226            0.5 * self.progress(1.0),
227            bottom_center,
228        )
229    }

231    /// Returns whether the shulker box can open
232    #[must_use]
233    pub fn can_open(&self, state: BlockStateId, world: &Arc<World>, pos: BlockPos) -> bool {
234        if !matches!(self.animation_status(), AnimationStatus::Closed) {
235            return true;
236        }
237
238        let direction = state.get_value(ShulkerBoxBlock::FACING);
239
240        let lid_open_bounding_box: WorldAabb = Self::get_progress_delta_aabb(
241            1.0,
242            direction,
243            0.0,
244            0.5,
245            DVec3::from(pos.get_bottom_center()),
246        )
247        .deflate(1.0E-6);
248
249        let collision = WorldCollisionProvider::new(world);
250        !collision.has_block_collision(&lid_open_bounding_box)
251    }

266    /// Interpolated progress for smooth rendering between ticks.
267    #[must_use]
268    pub fn progress(&self, partial_tick: f32) -> f32 {
269        let animation = self.animation.lock();
270        let new = animation.progress();
271        let old = animation.old_progress();
272        old + partial_tick * (new - old)
273    }

317    fn trigger_event(&self, kind: i32, data: i32) -> bool {
318        if kind == 1 {
319            self.open_count.store(data, Ordering::Relaxed);
320            if data == 0 {
321                self.animation.lock().animation_status = AnimationStatus::Closing;
322            }
323
324            if data == 1 {
325                self.animation.lock().animation_status = AnimationStatus::Opening;
326            }
327
328            true
329        } else {
330            false
331        }
332    }

363mod tests {
364    use steel_registry::{init_vanilla_registry, vanilla_blocks, vanilla_items};

366    use super::*;
367    use crate::test_support::fresh_test_world;
368
369    #[test]
370    fn applying_item_components_fills_the_slots_and_collects_them_back() {
371        init_vanilla_registry();
372        let world_fixture = fresh_test_world("shulker_box_apply_components");
373        let world = &world_fixture.world;
374        let shulker = ShulkerBoxBlockEntity::new(
375            Arc::downgrade(world),
376            BlockPos::new(8, 64, 8),
377            vanilla_blocks::SHULKER_BOX.default_state(),
378        );
379        let diamonds =
380            ItemStack::with_count_and_patch(&vanilla_items::DIAMOND, 5, DataComponentPatch::new());
381        let mut slots = vec![None; SHULKER_BOX_SLOTS];
382        slots[3] = Some(ItemStackTemplate::from_stack(&diamonds).expect("a plain stack converts"));
383        let contents = ItemContainerContents::new(slots).expect("27 slots fit the container limit");
384        let mut stack = ItemStack::new(&vanilla_items::SHULKER_BOX);
385        stack.set(CONTAINER, contents.clone());
386
387        shulker.apply_components_from_item_stack(&stack);
388
389        {
390            let container = shulker.storage.container.lock();
391            assert!(container.items().get(3).is(&vanilla_items::DIAMOND));
392            assert_eq!(container.items().get(3).count(), 5);
393            assert!(
394                container
395                    .items()
396                    .iter()
397                    .enumerate()
398                    .all(|(slot, item)| slot == 3 || item.is_empty())
399            );
400        }
401        assert!(
402            !shulker.base().stored_components().has(CONTAINER),
403            "an implicit component must not also be stored"
404        );
405        assert_eq!(
406            shulker
407                .collect_components()
408                .expect("valid components")
409                .get(CONTAINER),
410            Some(contents)
411        );
412    }
