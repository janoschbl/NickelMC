11/// Vanilla `Brain.SCHEDULE_UPDATE_DELAY`.
12const SCHEDULE_UPDATE_DELAY: i64 = 20;
13
14/// Memory states a brain requires before it switches to an activity.
15pub type ActivityRequirements = Vec<(MemoryModuleTypeRef, MemoryStatus)>;
16
17/// One activity's behaviors, requirements, and memories to erase.
18///
19/// Mirrors vanilla `ActivityData`.
20pub struct ActivityData {
21    pub(super) activity: ActivityRef,
22    pub(super) behaviors: Vec<(i32, Box<dyn BehaviorControl>)>,
23    pub(super) requirements: ActivityRequirements,
24    pub(super) memories_to_erase_when_stopped: Vec<MemoryModuleTypeRef>,
25}
26
27impl ActivityData {
28    /// Behaviors at consecutive priorities starting at `priority_of_first_behavior`.
29    #[must_use]
30    pub fn new(
31        activity: ActivityRef,
32        priority_of_first_behavior: i32,
33        behaviors: Vec<Box<dyn BehaviorControl>>,
34    ) -> Self {
35        let mut next_priority = priority_of_first_behavior;
36        let behaviors = behaviors
37            .into_iter()
38            .map(|behavior| {
39                let prioritized = (next_priority, behavior);
40                next_priority += 1;
41                prioritized
42            })
43            .collect();
44
45        Self::prioritized(activity, behaviors)
46    }
47
48    /// Behaviors at explicit priorities, which several behaviors may share.
49    #[must_use]
50    pub fn prioritized(
51        activity: ActivityRef,
52        behaviors: Vec<(i32, Box<dyn BehaviorControl>)>,
53    ) -> Self {
54        Self {
55            activity,
56            behaviors,
57            requirements: Vec::new(),
58            memories_to_erase_when_stopped: Vec::new(),
59        }
60    }
61
62    /// Memory states the brain requires before it will switch to this activity.
63    #[must_use]
64    pub fn requiring(mut self, requirements: ActivityRequirements) -> Self {
65        self.requirements = requirements;
66        self
67    }
68
69    /// Memories cleared when the brain switches away from this activity.
70    #[must_use]
71    pub fn erasing_when_stopped(mut self, memories: Vec<MemoryModuleTypeRef>) -> Self {
72        self.memories_to_erase_when_stopped = memories;
73        self
74    }
75}

77/// The activity state of a brain: what each activity needs, and which are running.
78///
79/// Activities are compared by the address of their registry entry, like
80/// [memories](Memories). Switching activity reads and erases memories, so those
81/// methods take the mob's [`Memories`]. Behaviors reach this through
82/// [`BrainContext`](super::BrainContext).
83pub struct Activities {
84    requirements: Vec<(ActivityRef, ActivityRequirements)>,
85    memories_to_erase_when_stopped: Vec<(ActivityRef, Vec<MemoryModuleTypeRef>)>,
86    core: Vec<ActivityRef>,
87    active: Vec<ActivityRef>,
88    default: ActivityRef,
89    last_schedule_update: i64,
90}
91
92impl Activities {
93    /// No registered activities, with the core activity set to
94    /// [`CORE`](vanilla_activities::CORE) and the default to
95    /// [`IDLE`](vanilla_activities::IDLE).
96    ///
97    /// Nothing is active until the brain performs its first activity switch.
98    #[must_use]
99    pub fn new() -> Self {
100        Self {
101            requirements: Vec::new(),
102            memories_to_erase_when_stopped: Vec::new(),
103            core: vec![&vanilla_activities::CORE],
104            active: Vec::new(),
105            default: &vanilla_activities::IDLE,
106            last_schedule_update: -9999,
107        }
108    }
109
110    /// Registers an activity's requirements and memories to erase, replacing any
111    /// registered earlier.
112    pub(super) fn register(
113        &mut self,
114        activity: ActivityRef,
115        requirements: ActivityRequirements,
116        memories_to_erase_when_stopped: Vec<MemoryModuleTypeRef>,
117    ) {
118        replace_entry(&mut self.requirements, activity, requirements);
119        if !memories_to_erase_when_stopped.is_empty() {
120            replace_entry(
121                &mut self.memories_to_erase_when_stopped,
122                activity,
123                memories_to_erase_when_stopped,
124            );
125        }
126    }
127
128    /// Sets the activities that stay active alongside the current one.
129    pub fn set_core_activities(&mut self, activities: Vec<ActivityRef>) {
130        self.core = activities;
131    }
132
133    /// Sets the activity the brain falls back to.
134    pub const fn set_default_activity(&mut self, activity: ActivityRef) {
135        self.default = activity;
136    }
137
138    /// Whether `activity` is currently active.
139    #[must_use]
140    pub fn is_active(&self, activity: ActivityRef) -> bool {
141        contains(&self.active, activity)
142    }
143
144    /// The active activity that is not a core one.
145    #[must_use]
146    pub fn active_non_core_activity(&self) -> Option<ActivityRef> {
147        self.active
148            .iter()
149            .copied()
150            .find(|&active| !contains(&self.core, active))
151    }
152
153    /// Switches to the default activity.
154    pub fn use_default_activity(&mut self, memories: &mut Memories) {
155        self.set_active_activity(memories, self.default);
156    }
157
158    /// Switches to `activity`, or to the default when its requirements fail.
159    pub fn set_active_activity_if_possible(
160        &mut self,
161        memories: &mut Memories,
162        activity: ActivityRef,
163    ) {
164        if self.requirements_are_met(memories, activity) {
165            self.set_active_activity(memories, activity);
166        } else {
167            self.use_default_activity(memories);
168        }
169    }
170
171    /// Switches to the first activity whose requirements are met.
172    pub fn set_active_activity_to_first_valid(
173        &mut self,
174        memories: &mut Memories,
175        activities: &[ActivityRef],
176    ) {
177        for &activity in activities {
178            if self.requirements_are_met(memories, activity) {
179                self.set_active_activity(memories, activity);
180                return;
181            }
182        }
183    }
184
185    /// Re-evaluates the scheduled activity at most every 20 ticks.
186    ///
187    /// `scheduled` yields the activity the mob's schedule asks for at its
188    /// position, and runs only when the throttle allows an update.
189    pub fn update_activity_from_schedule(
190        &mut self,
191        memories: &mut Memories,
192        game_time: i64,
193        scheduled: impl FnOnce() -> ActivityRef,
194    ) {
195        if game_time - self.last_schedule_update <= SCHEDULE_UPDATE_DELAY {
196            return;
197        }
198
199        self.last_schedule_update = game_time;
200        let scheduled = scheduled();
201        if !self.is_active(scheduled) {
202            self.set_active_activity_if_possible(memories, scheduled);
203        }
204    }
205
206    fn set_active_activity(&mut self, memories: &mut Memories, activity: ActivityRef) {
207        if self.is_active(activity) {
208            return;
209        }
210
211        self.erase_memories_for_other_activities_than(memories, activity);
212        self.active.clear();
213        self.active.extend(self.core.iter().copied());
214        self.active.push(activity);
215    }
216
217    /// Erases memories for activities that are no longer active.
218    fn erase_memories_for_other_activities_than(
219        &self,
220        memories: &mut Memories,
221        activity: ActivityRef,
222    ) {
223        for &stopped in &self.active {
224            if ptr::eq(stopped, activity) {
225                continue;
226            }
227
228            let to_erase = self
229                .memories_to_erase_when_stopped
230                .iter()
231                .find(|(registered, _)| ptr::eq(*registered, stopped));
232            for &memory in to_erase.into_iter().flat_map(|(_, memories)| memories) {
233                memories.erase(memory);
234            }
235        }
236    }
237
238    /// Whether the brain may switch to `activity`.
239    ///
240    /// An activity with no registered behaviors has no requirements entry and is
241    /// unreachable, so an unknown activity falls back to the default rather than
242    /// becoming silently available.
243    fn requirements_are_met(&self, memories: &Memories, activity: ActivityRef) -> bool {
244        let Some((_, requirements)) = self
245            .requirements
246            .iter()
247            .find(|(registered, _)| ptr::eq(*registered, activity))
248        else {
249            return false;
250        };
251
252        requirements
253            .iter()
254            .all(|&(memory, status)| memories.check(memory, status))
255    }
256}
257
258impl Default for Activities {
259    fn default() -> Self {
260        Self::new()
261    }
262}

264fn contains(activities: &[ActivityRef], activity: ActivityRef) -> bool {
265    activities.iter().any(|&other| ptr::eq(other, activity))
266}
267
268fn replace_entry<T>(entries: &mut Vec<(ActivityRef, T)>, activity: ActivityRef, value: T) {
269    match entries
270        .iter_mut()
271        .find(|(registered, _)| ptr::eq(*registered, activity))
272    {
273        Some((_, existing)) => *existing = value,
274        None => entries.push((activity, value)),
275    }
