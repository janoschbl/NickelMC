1//! The start/tick/stop lifecycle the brain drives.
2
3use super::Behavior;
4use crate::entity::ai::brain::context::BrainContext;
5use crate::entity::ai::brain::memory::MemoryModuleTypeRef;

7/// Whether a behavior is currently running.
8#[derive(Debug, Clone, Copy, PartialEq, Eq)]
9pub enum BehaviorStatus {
10    /// Not running.
11    Stopped,
12    /// Started and not yet stopped.
13    Running,
14}
15
16/// What the brain stores and drives.
17///
18/// Mirrors vanilla `BehaviorControl`.
19pub trait BehaviorControl: Send {
20    /// Current lifecycle state.
21    fn status(&self) -> BehaviorStatus;
22
23    /// Visits every memory this behavior needs, so the brain can register them.
24    fn visit_required_memories(&self, visit: &mut dyn FnMut(MemoryModuleTypeRef));
25
26    /// Starts the behavior if its conditions hold, returning whether it did.
27    fn try_start(&mut self, context: &mut BrainContext<'_>) -> bool;
28
29    /// Ticks the behavior, or stops it if it timed out or gave up.
30    fn tick_or_stop(&mut self, context: &mut BrainContext<'_>);
31
32    /// Stops the behavior unconditionally.
33    fn do_stop(&mut self, context: &mut BrainContext<'_>);
34
35    /// Name used in brain debug output.
36    fn debug_name(&self) -> &'static str;
37}
38
39/// A [`Behavior`] plus the lifecycle state vanilla keeps on `Behavior` itself.
40pub struct TimedBehavior<B: Behavior> {
41    behavior: B,
42    status: BehaviorStatus,
43    end_timestamp: i64,
44}
45
46impl<B: Behavior> TimedBehavior<B> {
47    /// Wraps `behavior`, stopped and not yet scheduled.
48    pub const fn new(behavior: B) -> Self {
49        Self {
50            behavior,
51            status: BehaviorStatus::Stopped,
52            end_timestamp: 0,
53        }
54    }
55
56    /// Vanilla `Behavior.timedOut`.
57    fn timed_out(&self, time: i64) -> bool {
58        self.behavior.timed_out(time, self.end_timestamp)
59    }
60
61    fn has_required_memories(&self, context: &BrainContext<'_>) -> bool {
62        self.behavior
63            .entry_condition()
64            .iter()
65            .all(|&(memory, status)| context.memories.check(memory, status))
66    }
67}
68
69impl<B: Behavior> BehaviorControl for TimedBehavior<B> {
70    fn status(&self) -> BehaviorStatus {
71        self.status
72    }
73
74    fn visit_required_memories(&self, visit: &mut dyn FnMut(MemoryModuleTypeRef)) {
75        for &(memory, _) in self.behavior.entry_condition() {
76            visit(memory);
77        }
78    }
79
80    /// Vanilla `Behavior.tryStart`.
81    fn try_start(&mut self, context: &mut BrainContext<'_>) -> bool {
82        if !self.has_required_memories(context)
83            || !self.behavior.check_extra_start_conditions(context)
84        {
85            return false;
86        }
87
88        self.status = BehaviorStatus::Running;
89        self.end_timestamp = context.time + i64::from(self.behavior.duration().roll());
90        self.behavior.start(context);
91        true
92    }
93
94    fn tick_or_stop(&mut self, context: &mut BrainContext<'_>) {
95        if self.timed_out(context.time) || !self.behavior.can_still_use(context) {
96            self.do_stop(context);
97            return;
98        }
99
100        self.behavior.tick(context);
101    }
102
103    fn do_stop(&mut self, context: &mut BrainContext<'_>) {
104        self.status = BehaviorStatus::Stopped;
105        self.behavior.stop(context);
106    }
107
108    fn debug_name(&self) -> &'static str {
109        self.behavior.debug_name()
110    }
111}

113/// Turns a [`Behavior`] into the [`BehaviorControl`] a brain stores.
114pub trait BehaviorExt: Behavior + Sized + 'static {
115    /// Wraps this behavior for registration on a brain.
116    fn control(self) -> Box<dyn BehaviorControl> {
117        Box::new(TimedBehavior::new(self))
118    }
119}
