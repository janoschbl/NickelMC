23/// How long one activation of a behavior may last.
24#[derive(Debug, Clone, Copy, PartialEq, Eq)]
25pub struct BehaviorDuration {
26    min: i32,
27    max: i32,
28}
29
30impl BehaviorDuration {
31    /// Vanilla's default 60 tick timeout.
32    pub const DEFAULT: Self = Self::fixed(60);
33
34    /// A duration of exactly `ticks`.
35    #[must_use]
36    pub const fn fixed(ticks: i32) -> Self {
37        Self::range(ticks, ticks)
38    }
39
40    /// A duration rolled from `min..=max` on every activation.
41    ///
42    /// # Panics
43    ///
44    /// Panics if `max` is below `min`.
45    #[must_use]
46    pub const fn range(min: i32, max: i32) -> Self {
47        assert!(min <= max, "behavior duration range is inverted");
48        Self { min, max }
49    }
50
51    /// Rolls one activation's length.
52    fn roll(self) -> i32 {
53        self.min + rand::random_range(0..=self.max - self.min)
54    }
55}
56
57impl Default for BehaviorDuration {
58    fn default() -> Self {
59        Self::DEFAULT
60    }
61}

63/// One thing a mob can do, gated by its memories.
64///
65/// Mirrors vanilla `Behavior`.
66pub trait Behavior: Send {
67    /// The memory states this behavior needs before it may start.
68    ///
69    /// Vanilla `Behavior.entryCondition`.
70    fn entry_condition(&self) -> &[(MemoryModuleTypeRef, MemoryStatus)] {
71        &[]
72    }
73
74    /// How long one activation lasts, rolled anew each time it starts.
75    fn duration(&self) -> BehaviorDuration {
76        BehaviorDuration::DEFAULT
77    }
78
79    /// Vanilla `Behavior.checkExtraStartConditions`.
80    fn check_extra_start_conditions(&mut self, _context: &mut BrainContext<'_>) -> bool {
81        true
82    }
83
84    /// Runs once when the behavior starts.
85    fn start(&mut self, _context: &mut BrainContext<'_>) {}
86
87    /// Whether the behavior should keep running.
88    fn can_still_use(&mut self, _context: &mut BrainContext<'_>) -> bool {
89        false
90    }
91
92    /// Runs every tick while the behavior is running.
93    fn tick(&mut self, _context: &mut BrainContext<'_>) {}
94
95    /// Runs once when the behavior stops, however it stopped.
96    fn stop(&mut self, _context: &mut BrainContext<'_>) {}
97
98    /// Override for behaviors whose lifetime is not limited by their duration.
99    fn timed_out(&self, time: i64, end_timestamp: i64) -> bool {
100        time > end_timestamp
101    }
102
103    /// Name used in brain debug output. Vanilla `Behavior.debugString`.
104    fn debug_name(&self) -> &'static str {
105        type_name::<Self>()
106    }
107}
