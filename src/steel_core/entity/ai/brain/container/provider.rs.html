1//! Building a mob's brain from its type's declarations and its saved memories.
2
3use super::{ActivityData, Brain};
4use crate::entity::Entity;
5use crate::entity::ai::brain::memory::{MemoryModuleTypeRef, PackedMemories};
6use crate::entity::ai::brain::sensor::SensorTypeRef;
7
8/// Creates the activities a mob's brain runs.
9pub type ActivitySupplier = fn(&dyn Entity) -> Vec<ActivityData>;
10
11/// What an entity type declares about its brain: the memories it holds, the
12/// sensors it runs, and the activities it supplies.
13///
14/// Every mob gets a fresh brain from [`make_brain`](Self::make_brain), so
15/// sensors and behaviors never share state between mobs.
16pub struct BrainProvider {
17    memory_types: Vec<MemoryModuleTypeRef>,
18    sensor_types: Vec<SensorTypeRef>,
19    activities: ActivitySupplier,
20}
21
22impl BrainProvider {
23    /// Declares a brain of `sensor_types` running the activities `activities`
24    /// builds for each mob.
25    ///
26    /// Sensors and behaviors register the memories they need on their own, so
27    /// `memory_types` only lists the ones nothing else declares.
28    #[must_use]
29    pub const fn new(
30        memory_types: Vec<MemoryModuleTypeRef>,
31        sensor_types: Vec<SensorTypeRef>,
32        activities: ActivitySupplier,
33    ) -> Self {
34        Self {
35            memory_types,
36            sensor_types,
37            activities,
38        }
39    }
40
41    /// Builds a brain for `mob`, restoring `packed` into its memories.
42    #[must_use]
43    pub fn make_brain(&self, mob: &dyn Entity, packed: &PackedMemories) -> Brain {
44        let mut brain = Brain::new();
45
46        for &memory in &self.memory_types {
47            brain.register_memory(memory);
48        }
49        for (index, sensor_type) in self.sensor_types.iter().enumerate() {
50            if !self.sensor_types[..index]
51                .iter()
52                .any(|previous| std::ptr::eq(*previous, *sensor_type))
53            {
54                brain.add_sensor(sensor_type.create());
55            }
56        }
57        for activity in (self.activities)(mob) {
58            brain.add_activity(activity);
59        }
60        brain.memories_mut().restore(packed);
61        brain
62    }
63}
