14/// A registered memory module type, erased of its value type.
15pub type MemoryModuleTypeRef = &'static MemoryModuleTypeEntry;

17/// Encodes and decodes one memory's value without knowing its Rust type.
18#[derive(Clone, Copy)]
19struct ErasedCodec {
20    encode: fn(&dyn MemoryValue) -> Option<NbtTag>,
21    decode: fn(&NbtTag) -> Option<Box<dyn MemoryValue>>,
22}
23
24impl ErasedCodec {
25    const fn of<V: PersistentMemoryValue + DowncastType>() -> Self {
26        Self {
27            encode: |value| value.downcast_ref::<V>().map(V::to_nbt),
28            decode: |tag| V::from_nbt(tag).map(|value| Box::new(value) as Box<dyn MemoryValue>),
29        }
30    }
31}

33/// The registry identity of one brain memory, without its value type.
34pub struct MemoryModuleTypeEntry {
35    /// Registry key identifying this memory.
36    pub key: Identifier,
37    value_type_key: DowncastTypeKey,
38    codec: Option<ErasedCodec>,
39}
40
41impl MemoryModuleTypeEntry {
42    /// Returns the downcast key of the Rust value this memory stores.
43    #[must_use]
44    pub const fn value_type_key(&self) -> DowncastTypeKey {
45        self.value_type_key
46    }
47
48    /// Returns whether this memory is saved with the mob.
49    #[must_use]
50    pub const fn can_serialize(&self) -> bool {
51        self.codec.is_some()
52    }
53
54    pub(super) fn encode(&self, value: &dyn MemoryValue) -> Option<NbtTag> {
55        (self.codec?.encode)(value)
56    }
57
58    pub(super) fn decode(&self, tag: &NbtTag) -> Option<Box<dyn MemoryValue>> {
59        (self.codec?.decode)(tag)
60    }
61}
62
63impl Debug for MemoryModuleTypeEntry {
64    fn fmt(&self, formatter: &mut Formatter<'_>) -> fmt::Result {
65        formatter
66            .debug_struct("MemoryModuleTypeEntry")
67            .field("key", &self.key)
68            .field("value_type_key", &self.value_type_key)
69            .field("can_serialize", &self.can_serialize())
70            .finish_non_exhaustive()
71    }
72}

74/// Typed handle for one brain memory.
75pub struct MemoryModuleType<V: MemoryValue + DowncastType> {
76    entry: MemoryModuleTypeEntry,
77    _marker: PhantomData<fn(&V)>,
78}
79
80impl<V: MemoryValue + DowncastType> MemoryModuleType<V> {
81    /// Creates a memory module type that stores `V` under `key` and is not saved.
82    #[must_use]
83    pub const fn new(key: Identifier) -> Self {
84        Self {
85            entry: MemoryModuleTypeEntry {
86                key,
87                value_type_key: V::TYPE_KEY,
88                codec: None,
89            },
90            _marker: PhantomData,
91        }
92    }
93
94    /// Creates a memory module type that stores `V` under `key` and is saved
95    /// with the mob.
96    #[must_use]
97    pub const fn new_persistent(key: Identifier) -> Self
98    where
99        V: PersistentMemoryValue,
100    {
101        Self {
102            entry: MemoryModuleTypeEntry {
103                key,
104                value_type_key: V::TYPE_KEY,
105                codec: Some(ErasedCodec::of::<V>()),
106            },
107            _marker: PhantomData,
108        }
109    }
110
111    /// Returns the erased registry entry for this memory.
112    #[must_use]
113    pub const fn entry(&'static self) -> MemoryModuleTypeRef {
114        &self.entry
115    }
116
117    /// Returns the registry key.
118    #[must_use]
119    pub const fn key(&self) -> &Identifier {
120        &self.entry.key
121    }
122}
123
124impl<V: MemoryValue + DowncastType> Debug for MemoryModuleType<V> {
125    fn fmt(&self, formatter: &mut Formatter<'_>) -> fmt::Result {
126        formatter
127            .debug_struct("MemoryModuleType")
128            .field("key", &self.entry.key)
129            .field("value_type_key", &self.entry.value_type_key)
130            .finish()
131    }
132}

134/// Registry of brain memory module types.
135pub struct MemoryModuleTypeRegistry {
136    memory_types: Vec<MemoryModuleTypeRef>,
137    by_key: FxHashMap<Identifier, usize>,
138    allows_registering: bool,
139}
140
141impl MemoryModuleTypeRegistry {
142    /// Creates an empty, still-writable registry.
143    #[must_use]
144    pub fn new() -> Self {
145        Self {
146            memory_types: Vec::new(),
147            by_key: FxHashMap::default(),
148            allows_registering: true,
149        }
150    }
151
152    /// Registers a memory module type and returns its registry id.
153    ///
154    /// # Panics
155    ///
156    /// Panics if the registry is frozen or the key is already registered.
157    pub fn register<V: MemoryValue + DowncastType>(
158        &mut self,
159        memory_type: &'static MemoryModuleType<V>,
160    ) -> usize {
161        self.register_entry(memory_type.entry())
162    }
163
164    fn register_entry(&mut self, entry: MemoryModuleTypeRef) -> usize {
165        assert!(
166            self.allows_registering,
167            "Cannot register memory module types after the registry has been frozen"
168        );
169        assert!(
170            !self.by_key.contains_key(&entry.key),
171            "Cannot register duplicate memory module type key: {}",
172            entry.key
173        );
174        let id = self.memory_types.len();
175        self.memory_types.push(entry);
176        self.by_key.insert(entry.key.clone(), id);
177        id
178    }
179
180    /// Rejects further registrations.
181    pub const fn freeze(&mut self) {
182        self.allows_registering = false;
183    }
184
185    /// Returns the memory module type registered under `key`.
186    #[must_use]
187    pub fn by_key(&self, key: &Identifier) -> Option<MemoryModuleTypeRef> {
188        self.by_key
189            .get(key)
190            .and_then(|&id| self.memory_types.get(id))
191            .copied()
192    }
193
194    /// Returns the memory module type registered under `id`.
195    #[must_use]
196    pub fn by_id(&self, id: usize) -> Option<MemoryModuleTypeRef> {
197        self.memory_types.get(id).copied()
198    }
199
200    /// Iterates registered memory module types in registration order.
201    pub fn iter(&self) -> impl Iterator<Item = MemoryModuleTypeRef> + '_ {
202        self.memory_types.iter().copied()
203    }
204
205    /// Returns the number of registered memory module types.
206    #[must_use]
207    pub const fn len(&self) -> usize {
208        self.memory_types.len()
209    }
210
211    /// Returns whether no memory module type is registered.
212    #[must_use]
213    pub const fn is_empty(&self) -> bool {
214        self.memory_types.is_empty()
215    }
216}
217
218impl Default for MemoryModuleTypeRegistry {
219    fn default() -> Self {
220        Self::new()
221    }
222}

224/// Wrapper for the global memory module type registry that implements `Deref`.
225pub struct MemoryModuleTypeRegistryLock(OnceLock<MemoryModuleTypeRegistry>);
226
227impl MemoryModuleTypeRegistryLock {
228    fn get_or_init(
229        &self,
230        init: impl FnOnce() -> MemoryModuleTypeRegistry,
231    ) -> &MemoryModuleTypeRegistry {
232        self.0.get_or_init(init)
233    }
234}
235
236impl Deref for MemoryModuleTypeRegistryLock {
237    type Target = MemoryModuleTypeRegistry;
238
239    fn deref(&self) -> &Self::Target {
240        self.0
241            .get()
242            .expect("Memory module type registry not initialized")
243    }
244}

246/// Global memory module type registry.
247///
248/// Access via deref: `MEMORY_MODULE_TYPES.by_key(&key)`.
249pub static MEMORY_MODULE_TYPES: MemoryModuleTypeRegistryLock =
250    MemoryModuleTypeRegistryLock(OnceLock::new());
251
252/// Initializes the global memory module type registry, once the main registry is
253/// frozen.
254pub(in crate::entity::ai::brain) fn init_memory_module_types() {
255    init_memory_module_types_with(|_| {});
256}
257
258pub(in crate::entity::ai::brain) fn init_memory_module_types_with(
259    register: impl FnOnce(&mut MemoryModuleTypeRegistry),
260) -> bool {
261    let mut initialized = false;
262    MEMORY_MODULE_TYPES.get_or_init(|| {
263        let mut registry = MemoryModuleTypeRegistry::new();
264        register_memory_module_types(&mut registry);
265        register(&mut registry);
266        registry.freeze();
267        initialized = true;
268        registry
269    });
270    initialized
271}
