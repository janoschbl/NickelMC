1//! Lazy visibility results cached for one sensor scan.
2
3use super::{Memories, MemoryValue, RememberedEntities};
4use crate::entity::SharedEntity;
5use rustc_hash::FxHashMap;
6use std::fmt;
7use steel_utils::{DowncastType, DowncastTypeKey, locks::SyncMutex};
8use uuid::Uuid;
9
10type VisibilityTest = Box<dyn Fn(&SharedEntity, &Memories) -> bool + Send + Sync>;
11
12pub struct NearestVisibleLivingEntities {
13    nearby: RememberedEntities,
14    test: VisibilityTest,
15    cache: SyncMutex<FxHashMap<Uuid, bool>>,
16}
17
18// SAFETY: Unique Steel-owned concrete payload key.
19unsafe impl DowncastType for NearestVisibleLivingEntities {
20    const TYPE_KEY: DowncastTypeKey = DowncastTypeKey::new("steel:memory_value/visible_entities");
21}
22impl MemoryValue for NearestVisibleLivingEntities {}
23impl fmt::Debug for NearestVisibleLivingEntities {
24    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
25        f.debug_struct("NearestVisibleLivingEntities")
26            .field("nearby", &self.nearby)
27            .finish_non_exhaustive()
28    }
29}
30impl NearestVisibleLivingEntities {
31    /// The empty visibility result used before a mob's first sensor scan.
32    pub fn empty() -> &'static Self {
33        static EMPTY: std::sync::LazyLock<NearestVisibleLivingEntities> =
34            std::sync::LazyLock::new(|| {
35                NearestVisibleLivingEntities::new(RememberedEntities::default(), |_, _| false)
36            });
37        &EMPTY
38    }
39    pub fn new(
40        nearby: RememberedEntities,
41        test: impl Fn(&SharedEntity, &Memories) -> bool + Send + Sync + 'static,
42    ) -> Self {
43        Self {
44            nearby,
45            test: Box::new(test),
46            cache: SyncMutex::default(),
47        }
48    }
49    pub fn nearby_entities(&self) -> &RememberedEntities {
50        &self.nearby
51    }
52    pub fn find_closest(
53        &self,
54        memories: &Memories,
55        filter: impl Fn(&SharedEntity) -> bool,
56    ) -> Option<SharedEntity> {
57        self.nearby
58            .resolve()
59            .find(|entity| filter(entity) && self.is_visible(entity, memories))
60    }
61    pub fn contains(&self, target: &SharedEntity, memories: &Memories) -> bool {
62        self.nearby
63            .resolve()
64            .any(|entity| std::sync::Arc::ptr_eq(&entity, target))
65            && self.is_visible(target, memories)
66    }
67    fn is_visible(&self, entity: &SharedEntity, memories: &Memories) -> bool {
68        if let Some(visible) = self.cache.lock().get(&entity.uuid()).copied() {
69            return visible;
70        }
71        let visible = (self.test)(entity, memories);
72        self.cache.lock().insert(entity.uuid(), visible);
73        visible
74    }
75}
