18pub struct PlayerSensor;
19impl Sensor for PlayerSensor {
20    fn requires(&self) -> &[MemoryModuleTypeRef] {
21        super::vanilla_sensors::NEAREST_PLAYERS_MEMORIES
22    }
23    fn do_tick(&mut self, context: &mut BrainContext<'_>) {
24        let Some(body) = context.mob.as_living_entity() else {
25            return;
26        };
27        let range = targeting::follow_range(body);
28        let mut players: Vec<crate::entity::SharedEntity> = Vec::new();
29        context.level.players.iter_players(|_, player| {
30            if !player.is_spectator()
31                && body.position().distance_squared(player.position()) < range * range
32            {
33                let entity: crate::entity::SharedEntity = player.clone();
34                players.push(entity);
35            }
36            true
37        });
38        players.sort_by(|a, b| {
39            body.position()
40                .distance_squared(a.position())
41                .total_cmp(&body.position().distance_squared(b.position()))
42        });
43        let visible: Vec<_> = players
44            .iter()
45            .filter(|player| {
46                player
47                    .as_living_entity()
48                    .is_some_and(|living| targeting::is_entity_targetable(context, living))
49            })
50            .collect();
51        let attackable: Vec<_> = visible
52            .iter()
53            .copied()
54            .filter(|player| {
55                player
56                    .as_living_entity()
57                    .is_some_and(|living| targeting::is_entity_attackable(context, living))
58            })
59            .collect();
60        let nearest = visible.first().map(|entity| RememberedEntity::new(entity));
61        let nearest_attackable = attackable
62            .first()
63            .map(|entity| RememberedEntity::new(entity));
64        let attackable: RememberedEntities = attackable.into_iter().collect();
65        context
66            .memories
67            .set(&memories::NEAREST_PLAYERS, players.iter().collect());
68        context
69            .memories
70            .set_optional(&memories::NEAREST_VISIBLE_PLAYER, nearest);
71        context.memories.set_optional(
72            &memories::NEAREST_VISIBLE_ATTACKABLE_PLAYER,
73            nearest_attackable,
74        );
75        context
76            .memories
77            .set(&memories::NEAREST_VISIBLE_ATTACKABLE_PLAYERS, attackable);
78    }
79}

81pub struct NearestLivingEntitySensor;
82impl Sensor for NearestLivingEntitySensor {
83    fn requires(&self) -> &[MemoryModuleTypeRef] {
84        super::vanilla_sensors::NEAREST_LIVING_ENTITIES_MEMORIES
85    }
86    fn do_tick(&mut self, context: &mut BrainContext<'_>) {
87        let Some(body) = context.mob.as_living_entity() else {
88            return;
89        };
90        let range = targeting::follow_range(body);
91        let mut nearby = context.level.get_entities_in_aabb_matching(
92            &body.bounding_box().inflate(range),
93            |entity| {
94                entity.uuid() != body.uuid()
95                    && entity
96                        .as_living_entity()
97                        .is_some_and(LivingEntity::is_alive)
98            },
99        );
100        nearby.sort_by(|a, b| {
101            body.position()
102                .distance_squared(a.position())
103                .total_cmp(&body.position().distance_squared(b.position()))
104        });
105        let weak_body = context
106            .level
107            .get_entity_by_id(body.id())
108            .map(|entity| Arc::downgrade(&entity));
109        let weak_level = Arc::downgrade(context.level);
110        let visible = NearestVisibleLivingEntities::new(
111            nearby.iter().collect(),
112            move |target, current_memories| {
113                let Some(body) = weak_body.as_ref().and_then(std::sync::Weak::upgrade) else {
114                    return false;
115                };
116                let Some(level) = weak_level.upgrade() else {
117                    return false;
118                };
119                let Some(living) = body.as_living_entity() else {
120                    return false;
121                };
122                let Some(target) = target.as_living_entity() else {
123                    return false;
124                };
125                let mut conditions =
126                    TargetingConditions::for_non_combat().range(targeting::follow_range(living));
127                if current_memories
128                    .get_internal(&memories::ATTACK_TARGET)
129                    .flatten()
130                    .is_some_and(|remembered| remembered.uuid == target.uuid())
131                {
132                    conditions = conditions.ignore_invisibility_testing();
133                }
134                conditions.test(level.as_ref(), Some(living), target)
135            },
136        );
137        context
138            .memories
139            .set(&memories::NEAREST_LIVING_ENTITIES, nearby.iter().collect());
140        context
141            .memories
142            .set(&memories::NEAREST_VISIBLE_LIVING_ENTITIES, visible);
143    }
144}

146pub struct HurtBySensor;
147impl Sensor for HurtBySensor {
148    fn requires(&self) -> &[MemoryModuleTypeRef] {
149        super::vanilla_sensors::HURT_BY_MEMORIES
150    }
151    fn do_tick(&mut self, context: &mut BrainContext<'_>) {
152        let Some(body) = context.mob.as_living_entity() else {
153            return;
154        };
155        if let Some(source) = body.last_damage_source() {
156            context
157                .memories
158                .set(&memories::HURT_BY, RememberedDamage::from(&source));
159            if let Some(entity) = source
160                .causing_entity()
161                .filter(|entity| entity.as_living_entity().is_some())
162            {
163                context
164                    .memories
165                    .set(&memories::HURT_BY_ENTITY, RememberedEntity::new(entity));
166            }
167        } else {
168            context.memories.erase(memories::HURT_BY.entry());
169        }
170        if context
171            .memories
172            .get_internal(&memories::HURT_BY_ENTITY)
173            .flatten()
174            .is_some_and(|entity| {
175                entity.resolve().is_none_or(|entity| {
176                    !entity
177                        .as_living_entity()
178                        .is_some_and(LivingEntity::is_alive)
179                        || entity
180                            .level()
181                            .is_none_or(|level| !Arc::ptr_eq(&level, context.level))
182                })
183            })
184        {
185            context.memories.erase(memories::HURT_BY_ENTITY.entry());
186        }
187    }
188}
