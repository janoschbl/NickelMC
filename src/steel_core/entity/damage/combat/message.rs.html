16fn translated(key: String, args: Vec<TextComponent>) -> TextComponent {
17    TranslatedMessage {
18        key: key.into(),
19        fallback: None,
20        args: Some(args.into_boxed_slice()),
21    }
22    .component()
23}

25impl CombatTracker {
26    pub fn death_message(&self, victim: &dyn LivingEntity) -> TextComponent {
27        let Some(killing_blow) = self.entries.last() else {
28            return translated(
29                "death.attack.generic".to_owned(),
30                vec![victim.display_name()],
31            );
32        };
33        let source = &killing_blow.source;
34        match source.damage_type.death_message_type {
35            DeathMessageType::FallVariants => {
36                if let Some(entry) = self.most_significant_fall() {
37                    return Self::fall_message(
38                        victim,
39                        entry,
40                        source.causing_entity().map(AsRef::as_ref),
41                    );
42                }
43            }
44            DeathMessageType::IntentionalGameDesign => {
45                let base = format!("death.attack.{}", source.damage_type.message_id);
46                let link = TextComponent::const_plain("[")
47                    .add_child(translated(format!("{base}.link"), vec![]))
48                    .add_child("]")
49                    .click_event(ClickEvent::open_url(
50                        "https://bugs.mojang.com/browse/MCPE-28723",
51                    ))
52                    .hover_event(HoverEvent::show_text("MCPE-28723"));
53                return translated(format!("{base}.message"), vec![victim.display_name(), link]);
54            }
55            DeathMessageType::Default => {}
56        }
57        let key = format!("death.attack.{}", source.damage_type.message_id);
58        if source.causing_entity().is_none() && source.direct_entity().is_none() {
59            let killer = victim
60                .level()
61                .and_then(|world| get_kill_credit(victim, &world));
62            return if let Some(killer) = killer {
63                translated(
64                    format!("{key}.player"),
65                    vec![victim.display_name(), killer.display_name()],
66                )
67            } else {
68                translated(key, vec![victim.display_name()])
69            };
70        }
71        let attacker = source.causing_entity().or(source.direct_entity());
72        let mut args = vec![victim.display_name()];
73        if let Some(attacker) = attacker {
74            args.push(attacker.display_name());
75        }
76        if let Some(item_name) = source
77            .causing_entity()
78            .and_then(|entity| Self::named_weapon(entity.as_ref()))
79        {
80            args.push(item_name);
81            return translated(format!("{key}.item"), args);
82        }
83        translated(key, args)
84    }

86    fn named_weapon(attacker: &dyn Entity) -> Option<TextComponent> {
87        let living = attacker.as_living_entity()?;
88        let mut result = None;
89        living.with_equipment_slot(EquipmentSlot::MainHand, &mut |item| {
90            if !item.is_empty() && item.has(CUSTOM_NAME) {
91                result = Some(ITEM_BEHAVIORS.display_name(item));
92            }
93        });
94        result
95    }

97    fn fall_message(
98        victim: &dyn LivingEntity,
99        entry: &CombatEntry,
100        killer: Option<&dyn Entity>,
101    ) -> TextComponent {
102        if entry.source.is(&DamageTypeTag::IS_FALL)
103            || entry
104                .source
105                .is(&DamageTypeTag::ALWAYS_MOST_SIGNIFICANT_FALL)
106        {
107            return translated(
108                format!(
109                    "death.fell.accident.{}",
110                    entry.fall_location.unwrap_or("generic")
111                ),
112                vec![victim.display_name()],
113            );
114        }
115        let attacker = entry.source.causing_entity();
116        let attacker_name = attacker.map(|entity| entity.display_name());
117        let killer_name = killer.map(Entity::display_name);
118        let (key, entity, name) = if attacker_name.is_some() && attacker_name != killer_name {
119            (
120                "death.fell.assist",
121                attacker.map(AsRef::as_ref),
122                attacker_name,
123            )
124        } else if killer_name.is_some() {
125            ("death.fell.finish", killer, killer_name)
126        } else {
127            return translated("death.fell.killer".to_owned(), vec![victim.display_name()]);
128        };
129        let mut args = vec![victim.display_name()];
130        if let Some(name) = name {
131            args.push(name);
132        }
133        if let Some(item_name) = entity.and_then(Self::named_weapon) {
134            args.push(item_name);
135            return translated(format!("{key}.item"), args);
136        }
137        translated(key.to_owned(), args)
138    }

140    fn most_significant_fall(&self) -> Option<&CombatEntry> {
141        let mut result = None;
142        let mut alternative = None;
143        let mut alt_damage = 0.0;
144        let mut best_fall = 0.0;
145        for (index, entry) in self.entries.iter().enumerate() {
146            let fake_fall = entry
147                .source
148                .is(&DamageTypeTag::ALWAYS_MOST_SIGNIFICANT_FALL);
149            let distance = if fake_fall {
150                f32::MAX
151            } else {
152                entry.fall_distance
153            };
154            if (entry.source.is(&DamageTypeTag::IS_FALL) || fake_fall)
155                && distance > 0.0
156                && (result.is_none() || distance > best_fall)
157            {
158                result = Some(&self.entries[index.saturating_sub(1)]);
159                best_fall = distance;
160            }
161            if entry.fall_location.is_some() && (alternative.is_none() || entry.damage > alt_damage)
162            {
163                alternative = Some(entry);
164                alt_damage = entry.damage;
165            }
166        }
167        if best_fall > 5.0 && result.is_some() {
168            result
169        } else if alt_damage > 5.0 {
170            alternative
171        } else {
172            None
173        }
174    }
