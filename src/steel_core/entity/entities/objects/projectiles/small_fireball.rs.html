29/// A small accelerating fireball.
30#[entity_behavior(class = "SmallFireball")]
31pub struct SmallFireballEntity {
32    base: EntityBase,
33    entity_type: EntityTypeRef,
34    entity_data: SyncMutex<SmallFireballEntityData>,
35    projectile_base: ProjectileBase,
36    hurting_base: HurtingProjectileBase,
37}
38
39// SAFETY: This owner-scoped key uniquely identifies SmallFireballEntity.
40unsafe impl DowncastType for SmallFireballEntity {
41    const TYPE_KEY: DowncastTypeKey = DowncastTypeKey::new("steel:entity/small_fireball");
42}
43
44impl SmallFireballEntity {
45    /// Creates an unowned small fireball.
46    pub fn new(entity_type: EntityTypeRef, id: i32, position: DVec3, world: Weak<World>) -> Self {
47        Self::with_base(
48            EntityBase::new(id, position, entity_type.dimensions, world),
49            entity_type,
50        )
51    }
52
53    /// Restores a saved small fireball.
54    pub fn from_saved(entity_type: EntityTypeRef, load: EntityBaseLoad) -> Self {
55        Self::with_base(
56            EntityBase::from_load(load, entity_type.dimensions),
57            entity_type,
58        )
59    }
60
61    fn with_base(base: EntityBase, entity_type: EntityTypeRef) -> Self {
62        let entity = Self {
63            base,
64            entity_type,
65            entity_data: SyncMutex::new(SmallFireballEntityData::new()),
66            projectile_base: ProjectileBase::new(),
67            hurting_base: HurtingProjectileBase::default(),
68        };
69        entity.set_item(ItemStack::new(&vanilla_items::FIRE_CHARGE));
70        entity
71    }
72}

120impl Fireball for SmallFireballEntity {
121    fn get_item(&self) -> ItemStack {
122        self.entity_data.lock().fireball.item_stack.get().clone()
123    }
124    fn set_item(&self, item: ItemStack) {
125        let item = if item.is_empty() {
126            ItemStack::new(&vanilla_items::FIRE_CHARGE)
127        } else {
128            item.copy_with_count(1)
129        };
130        self.entity_data.lock().fireball.item_stack.set(item);
131    }

134impl Projectile for SmallFireballEntity {
135    fn projectile_base(&self) -> &ProjectileBase {
136        &self.projectile_base
137    }
138    fn hurt(&self, _world: &World, _source: &DamageSource, _amount: f32) -> bool {
139        false
140    }
141    fn can_hit_entity(&self, entity: &dyn Entity) -> bool {
142        self.base_can_hit_entity(entity) && !entity.no_physics()
143    }
144    fn on_deflection(&self, by_attack: bool) {
145        self.hurting_on_deflection(by_attack);
146    }
147    fn on_hit_entity(self: Arc<Self>, entity: &SharedEntity, _location: DVec3) {
148        let Some(world) = self.level() else { return };
149        let remaining_fire = entity.remaining_fire_ticks();
150        entity.ignite_for_ticks(100);
151        let mut source = DamageSource::environment(&vanilla_damage_types::UNATTRIBUTED_FIREBALL)
152            .with_direct_entity(Arc::clone(&self) as SharedEntity);
153        if let Some(owner) = self.get_owner() {
154            source = DamageSource::environment(&vanilla_damage_types::FIREBALL)
155                .with_direct_entity(Arc::clone(&self) as SharedEntity)
156                .with_causing_entity(owner);
157        }
158        if !entity.hurt(&world, &source, 5.0) {
159            entity.set_remaining_fire_ticks(remaining_fire);
160        } else {
161            crate::enchantment_helper::do_post_attack_effects(&world, entity, &source);
162        }
163    }
164    fn on_hit_block(self: Arc<Self>, hit: &ClipHitResult) {
165        self.projectile_on_hit_block(hit);
166        let Some(world) = self.level() else { return };
167        if self
168            .get_owner()
169            .is_some_and(|owner| owner.as_mob().is_some())
170            && !world.get_game_rule(&vanilla_game_rules::MOB_GRIEFING)
171        {
172            return;
173        }
174        let pos = hit.block_pos.relative(hit.direction);
175        if world.get_block_state(pos).is_air() {
176            world.set_block(
177                pos,
178                FireBlock::get_state(world.as_ref(), pos),
179                UpdateFlags::UPDATE_ALL,
180            );
181        }
182    }
183    fn on_hit(self: Arc<Self>, hit: &ProjectileHit) {
184        Arc::clone(&self).projectile_on_hit(hit);
185        self.set_removed(RemovalReason::Discarded);
186    }
