1//! Multipart entity support.
2//!
3//! A multipart entity is one whose hitbox is split across several sub-entities, so
4//! that different regions can take different damage. The Ender Dragon is the only
5//! vanilla example.
6//!
7//! This version of vanilla has no shared base for those sub-entities:
8//! `EnderDragonPart extends Entity` directly, and every engine site that has to
9//! recognize one does so with `instanceof EnderDragonPart`. Steel instead follows
10//! `NeoForge`'s `PartEntity` shape, so world queries, damage redirection and the
11//! entity tracker never name a specific mob and a third-party multipart mob works
12//! without engine changes.
13//!
14//! Parts are deliberately **not** registered with the world's entity manager, which
15//! matches vanilla keeping them only in `ServerLevel.dragonParts`: they are never
16//! sectioned, never saved, and never sent to clients. The client synthesizes them
17//! from the parent's network ID; see [`crate::entity::EntityIdBlock`].
18
19use std::sync::{Arc, OnceLock, Weak};
20
21use glam::DVec3;
22use steel_registry::entity_type::EntityDimensions;

24use super::{Entity, SharedEntity, WeakEntity};

26/// A sub-entity owned by a multipart parent.
27///
28/// Mirrors vanilla `EnderDragonPart`, generalized so engine code can support any
29/// multipart mob. Implementors supply [`PartEntityBase`] through
30/// [`Self::part_base`] and get the rest for free.
31pub trait PartEntity: Entity {
32    /// Returns this part's shared vanilla state.
33    fn part_base(&self) -> &PartEntityBase;
34
35    /// Returns the entity this part belongs to.
36    ///
37    /// Mirrors vanilla `EnderDragonPart.parentMob`. `None` only before the owning
38    /// `Arc` has been bound, or once the parent has been dropped.
39    fn parent(&self) -> Option<SharedEntity> {
40        self.part_base().parent()
41    }
42
43    /// Returns this part's vanilla name, such as `"head"`, `"wing"` or `"tail"`.
44    ///
45    /// Mirrors vanilla `EnderDragonPart.name`. Parts of the same kind share a name.
46    fn part_name(&self) -> &'static str {
47        self.part_base().name()
48    }
49
50    /// Moves this part to `position`.
51    ///
52    /// Mirrors vanilla `EnderDragon.tickPart`. Parts live outside the world entity
53    /// manager, so this bypasses the manager-commit path that a tracked entity's
54    /// movement goes through.
55    fn set_part_position(&self, position: DVec3) {
56        self.base().set_position_local(position);
57    }
58}
59
60/// Shared state for a [`PartEntity`].
61///
62/// Mirrors the `parentMob`, `name` and `size` fields of vanilla `EnderDragonPart`.
63#[derive(Debug)]
64pub struct PartEntityBase {
65    /// The owning entity.
66    ///
67    /// Vanilla passes `this` from the parent's constructor. Rust cannot hand out a
68    /// reference to a value that is still being constructed, so the world binds this
69    /// once the parent's `Arc` exists and it starts being tracked.
70    parent: OnceLock<WeakEntity>,
71    name: &'static str,
72    size: EntityDimensions,
73}
74
75impl PartEntityBase {
76    /// Creates part state with the vanilla name and hitbox size.
77    #[must_use]
78    pub const fn new(name: &'static str, size: EntityDimensions) -> Self {
79        Self {
80            parent: OnceLock::new(),
81            name,
82            size,
83        }
84    }
85
86    /// Returns the owning entity, if it is still alive and already bound.
87    #[must_use]
88    pub fn parent(&self) -> Option<SharedEntity> {
89        self.parent.get().and_then(Weak::upgrade)
90    }
91
92    /// Returns the part's vanilla name.
93    #[must_use]
94    pub const fn name(&self) -> &'static str {
95        self.name
96    }
97
98    /// Returns the part's hitbox size.
99    ///
100    /// Mirrors vanilla `EnderDragonPart.getDimensions`, which ignores the pose.
101    #[must_use]
102    pub const fn size(&self) -> EntityDimensions {
103        self.size
104    }
105
106    /// Binds the owning entity.
107    ///
108    /// Idempotent: later calls are ignored, so re-tracking a parent cannot rebind
109    /// its parts to a different owner.
110    pub fn bind_parent(&self, parent: &SharedEntity) {
111        let _ = self.parent.set(Arc::downgrade(parent));
112    }
113}
