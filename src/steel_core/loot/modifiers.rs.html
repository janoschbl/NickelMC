10pub(super) fn set_enchantment(item: &mut ItemStack, key: Identifier, level: i32, add: bool) {
11    let Some(enchantments) = item.get_enchantments_for_crafting() else {
12        return;
13    };
14    let base = if add {
15        enchantments.get_level(&key) as i32
16    } else {
17        0
18    };
19    item.set_enchantments(
20        &[(key, base.wrapping_add(level).clamp(0, 255) as u32)],
21        false,
22    );
23}

25pub(super) fn smelt(item: &mut ItemStack, use_input_count: bool) {
26    if item.is_empty() {
27        return;
28    }
29    let input = SingleItemRecipeInput::new(item.copy_with_count(item.count()));
30    let Some(recipe) = REGISTRY
31        .recipes
32        .find_match(&vanilla_recipe_types::SMELTING, &input)
33    else {
34        log::warn!(
35            "Couldn't smelt {} because there is no smelting recipe",
36            item.item.key
37        );
38        return;
39    };
40    let mut result = recipe.data().result.create();
41    if result.is_empty() {
42        return;
43    }
44    let count = if use_input_count { item.count() } else { 1 };
45    result.set_count(
46        count
47            .wrapping_mul(result.count())
48            .min(result.max_stack_size()),
49    );
50    *item = result;
51}

53pub(super) fn uniform_bonus(
54    item: &mut ItemStack,
55    random: &mut RandomSource,
56    level: i32,
57    multiplier: i32,
58) {
59    let bonus = random.next_i32_bounded(multiplier * level + 1);
60    item.set_count(item.count().wrapping_add(bonus));
61}

63// Widen before adding the half: an f32 addition would round the value directly
64// below 0.5 up to 1.0. Java rounds negative ties toward positive infinity.
65pub(super) fn round_count(value: f32) -> i32 {
66    (f64::from(value) + 0.5).floor() as i32
