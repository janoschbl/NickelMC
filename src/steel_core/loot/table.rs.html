7/// A registered loot table's identity and execution metadata.
8#[derive(Debug)]
9pub struct LootTable {
10    /// Registry identity used for saves and compiled implementations.
11    pub key: Identifier,
12    /// Vanilla context parameter-set identity.
13    pub param_set: Identifier,
14    /// Optional named random sequence used by world-aware callers.
15    pub random_sequence: Option<Identifier>,
16}
17
18/// A stable reference to a registered loot table.
19pub type LootTableRef = &'static LootTable;
20
21/// Vanilla `LootTable.EMPTY`: what a saved table key resolves to when no table is registered for it.
22pub static EMPTY_LOOT_TABLE: LootTable = LootTable {
23    key: Identifier::new_static("steel", "empty"),
24    param_set: Identifier::vanilla_static("empty"),
25    random_sequence: None,
26};

40impl LootTable {
41    /// Streams raw stacks. A failing callable may already have emitted output.
42    /// Use `get_random_items` to collect before publishing gameplay output.
43    pub fn emit_raw(
44        &self,
45        context: &mut LootContext<'_>,
46        output: &mut LootOutput<'_>,
47    ) -> LootResult<()> {
48        if context.visited.contains(&self.key) {
49            log::warn!("Detected infinite loop in loot table {}", self.key);
50            return Ok(());
51        }
52        let registry = context.tables.unwrap_or(&VANILLA_TABLES);
53        let Some(function) = registry.functions.get(&self.key).copied() else {
54            return Err(LootError::MissingTable(self.key.clone()));
55        };
56        context.visited.push(self.key.clone());
57        let result = function(context, output);
58        context.visited.pop();
59        result
60    }
61
62    /// Collects raw output without normalizing stack sizes.
63    pub fn get_random_items_raw(
64        &self,
65        context: &mut LootContext<'_>,
66    ) -> LootResult<Vec<ItemStack>> {
67        let mut items = Vec::new();
68        self.emit_raw(context, &mut |_, item| {
69            items.push(item);
70            Ok(())
71        })?;
72        Ok(items)
73    }
74
75    /// Collects and splits output into legal item stack sizes.
76    ///
77    /// Enabled-feature filtering from vanilla's createStackSplitter is deferred:
78    /// World has no enabled-feature contract and item definitions carry no required
79    /// feature sets. This does not claim parity for feature-restricted items.
80    pub fn get_random_items(&self, context: &mut LootContext<'_>) -> LootResult<Vec<ItemStack>> {
81        let mut items = Vec::new();
82        self.emit_raw(context, &mut |_, mut item| {
83            // Vanilla forwards sub-maximal stacks, including empty ones. Fill
84            // removes empties later, after collecting and shuffling available slots.
85            let max = item.max_stack_size();
86            if item.count() < max {
87                items.push(item);
88            } else {
89                while item.count() > 0 {
90                    items.push(item.split(max));
91                }
92            }
93            Ok(())
94        })?;
95        Ok(items)
96    }
97
98    /// Generates all loot before acquiring a destination container's lock.
99    pub fn generate_for_container<'a>(
100        &self,
101        context: &'a mut LootContext<'_>,
102    ) -> LootResult<GeneratedLoot<'a>> {
103        let items = self.get_random_items(context)?;
104        Ok(GeneratedLoot::new(items, context.rng))
105    }
106}
